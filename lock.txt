# Create Policy Keys for Edge & Chrome
$edgePath = "HKLM:\SOFTWARE\Policies\Microsoft\Edge"
$chromePath = "HKLM:\SOFTWARE\Policies\Google\Chrome"
New-Item -Path $edgePath -Force | Out-Null
New-Item -Path $chromePath -Force | Out-Null

# 1. Disable Incognito / InPrivate Browsing (Students cannot hide history)
Set-ItemProperty -Path $edgePath -Name "InPrivateModeAvailability" -Value 1 -Type DWord
Set-ItemProperty -Path $chromePath -Name "IncognitoModeAvailability" -Value 1 -Type DWord

# 2. Force SafeSearch on Google / Bing / YouTube
Set-ItemProperty -Path $edgePath -Name "ForceGoogleSafeSearch" -Value 1 -Type DWord
Set-ItemProperty -Path $edgePath -Name "ForceYouTubeRestrict" -Value 2 -Type DWord
Set-ItemProperty -Path $chromePath -Name "ForceGoogleSafeSearch" -Value 1 -Type DWord
Set-ItemProperty -Path $chromePath -Name "ForceYouTubeRestrict" -Value 2 -Type DWord

# 3. Block Specific URLs (e.g., Social Media, Gaming, Streaming)
$edgeBlocklist = "$edgePath\URLBlocklist"
New-Item -Path $edgeBlocklist -Force | Out-Null
Set-ItemProperty -Path $edgeBlocklist -Name "1" -Value "*facebook.com*"
Set-ItemProperty -Path $edgeBlocklist -Name "2" -Value "*instagram.com*"
Set-ItemProperty -Path $edgeBlocklist -Name "3" -Value "*tiktok.com*"
Set-ItemProperty -Path $edgeBlocklist -Name "4" -Value "*roblox.com*"
Set-ItemProperty -Path $edgeBlocklist -Name "5" -Value "*poki.com*"
